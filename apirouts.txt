CookMate API routes
Base URL: http://localhost:3000
Routes marked [AUTH] need the header: Authorization: Bearer <token>


HEALTH
GET    /health                      // check the server is running


AUTH
POST   /api/auth/signup             // create an account          body: { "name": "Jane", "email": "jane@example.com", "password": "password123" }
POST   /api/auth/login              // log in, returns a token    body: { "email": "jane@example.com", "password": "password123" }
GET    /api/auth/me                 // get the logged-in user     [AUTH]


RECIPES
GET    /api/recipes                 // list all my recipes        [AUTH]
GET    /api/recipes/latest          // my newest recipe           [AUTH]
GET    /api/recipes/find?id=ID      // one recipe by id           [AUTH]
POST   /api/recipes/ingredients     // AI ingredients for a meal  [AUTH]  body: { "mealName": "Jollof rice" }
POST   /api/recipes/generate        // AI recipe, saved           [AUTH]  body: { "mealName": "Jollof rice", "availableIngredients": ["rice", "tomato"], "excludeCommonAllergens": false, "quickMealOnly": true }
POST   /api/recipes/suggest         // AI meal ideas              [AUTH]  body: { "availableIngredients": ["chicken", "rice"] }


ALLERGENS
GET    /api/allergens               // list my allergen foods     [AUTH]
POST   /api/allergens               // add several foods          [AUTH]  body: { "allergens": ["peanut", "shrimp"] }
POST   /api/allergens/foods         // add one food               [AUTH]  body: { "name": "milk" }
DELETE /api/allergens/FOOD_ID       // remove a food from my list [AUTH]
